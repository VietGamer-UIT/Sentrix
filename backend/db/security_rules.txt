// NGUYÊN LÝ HOẠT ĐỘNG — ĐỌC KỸ TRƯỚC KHI SỬA
// ===========================================================================
//
// Sentrix dùng Firebase Authentication kết hợp Custom Claims để kiểm soát
// quyền truy cập theo từng tenant.
//
// Khi một doanh nghiệp đăng ký Sentrix, backend sẽ tạo tài khoản Firebase Auth
// cho chủ doanh nghiệp và gắn vào token 1 Custom Claim:
//
//   { "tenant_id": "pho-ba-lan_1722500000000" }
//
// Custom Claim này được gắn bởi Firebase Admin SDK (phía backend Tuyền làm),
// KHÔNG phải do client tự set — client không thể giả mạo claim này.
//
// Khi chủ doanh nghiệp đăng nhập vào Dashboard và gọi Firestore:
//   request.auth.token.tenant_id == "pho-ba-lan_1722500000000"
//
// Security Rule bên dưới kiểm tra: claim trong token có khớp với tenantId
// trong PATH không? Nếu không khớp → TỪ CHỐI (deny).
//
// Kết quả: Quán A chỉ đọc/ghi được path tenants/id-quan-A/**
//          Quán B không thể đọc gì trong tenants/id-quan-A/** dù cố tình.
//          Không phụ thuộc vào lập trình viên có "nhớ filter" hay không.
//
// ===========================================================================

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // -----------------------------------------------------------------------
    // QUY TẮC GỐC: TỪ CHỐI TẤT CẢ — trừ những gì được phép tường minh bên dưới
    // -----------------------------------------------------------------------
    match /{document=**} {
      allow read, write: if false;
    }

    // -----------------------------------------------------------------------
    // TENANT DATA — Mỗi tenant chỉ đọc/ghi được "phân vùng" của mình
    // -----------------------------------------------------------------------
    match /tenants/{tenantId} {
      // Chủ doanh nghiệp đọc được thông tin tenant của mình (tên quán, gói dịch vụ...)
      allow read: if request.auth != null
                  && request.auth.token.tenant_id == tenantId;

      // Chỉ backend (Service Account không có tenant_id claim) mới được tạo/sửa tenant
      // Client không bao giờ được write trực tiếp vào document tenant
      allow write: if false;

      // Sub-collections: feedbacks và customers
      match /{subcollection}/{docId} {
        // Chủ doanh nghiệp đọc/ghi được mọi sub-collection trong tenant của mình
        allow read, write: if request.auth != null
                           && request.auth.token.tenant_id == tenantId;
      }

      // Wildcard đệ quy: bao phủ mọi sub-collection lồng nhau (nếu sau này thêm)
      match /{subcollection}/{docId}/{nested=**} {
        allow read, write: if request.auth != null
                           && request.auth.token.tenant_id == tenantId;
      }
    }

    // -----------------------------------------------------------------------
    // WEB CLIENT (khách hàng gửi feedback qua QR) — KHÔNG dùng Firestore trực tiếp
    // -----------------------------------------------------------------------
    // Khách hàng quét QR → gọi FastAPI backend → backend ghi Firestore bằng Admin SDK.
    // Admin SDK BYPASS hoàn toàn Security Rules → không cần thêm rule cho client.
    // Web client của Việt KHÔNG kết nối Firestore trực tiếp — chỉ gọi REST API.
    //
    // Nếu sau này cần web client đọc Firestore trực tiếp (ví dụ realtime preview),
    // cần thêm rule riêng ở đây và báo Tuyền để cập nhật.
    // -----------------------------------------------------------------------
  }
}

// ===========================================================================
// KIỂM TRA RULE BẰNG FIRESTORE EMULATOR (khuyến nghị trước khi deploy thật)
// ===========================================================================
//
// 1. Cài Firebase CLI: npm install -g firebase-tools
// 2. firebase init emulators  (chọn Firestore)
// 3. firebase emulators:start
// 4. Vào http://localhost:4000 → Rules Playground để test các case:
//
// CASE 1 (phải PASS): Tenant A đọc tenants/id-A/feedbacks/xyz
//   Auth: { uid: "user-A", token: { tenant_id: "id-A" } }
//   Path: /tenants/id-A/feedbacks/xyz
//   Operation: read → Expected: ALLOW
//
// CASE 2 (phải FAIL): Tenant A đọc data của Tenant B
//   Auth: { uid: "user-A", token: { tenant_id: "id-A" } }
//   Path: /tenants/id-B/feedbacks/xyz
//   Operation: read → Expected: DENY
//
// CASE 3 (phải FAIL): Không đăng nhập → đọc bất kỳ data
//   Auth: không có (unauthenticated)
//   Path: /tenants/id-A/feedbacks/xyz
//   Operation: read → Expected: DENY
//
// CASE 4 (phải FAIL): Client cố ghi trực tiếp vào document tenant
//   Auth: { uid: "user-A", token: { tenant_id: "id-A" } }
//   Path: /tenants/id-A
//   Operation: write → Expected: DENY (chỉ Admin SDK mới ghi được)
